package ru.kode.test.app.schema

import kotlin.Any
import kotlin.Boolean
import kotlin.collections.List
import kotlin.collections.Map
import ru.kode.way.Event
import ru.kode.way.Path
import ru.kode.way.RegionId
import ru.kode.way.Schema
import ru.kode.way.Segment

public class Ph01tabsSchema : Schema {
  override val rootSegment: Segment =
      Segment("ph01tabs@ParallelHistory:src/test/resources/schema-parallel-history.dot")

  override val childSchemas: Map<Segment, Schema> =
      mapOf(Segment("ph01left@ParallelHistory:src/test/resources/schema-parallel-history.dot") to Ph01leftSchema(), Segment("ph01right@ParallelHistory:src/test/resources/schema-parallel-history.dot") to Ph01rightSchema())

  override val regions: List<RegionId> =
      listOf(RegionId(Path(listOf(Segment("ph01tabs@ParallelHistory:src/test/resources/schema-parallel-history.dot"), Segment("ph01left@ParallelHistory:src/test/resources/schema-parallel-history.dot")))), RegionId(Path(listOf(Segment("ph01tabs@ParallelHistory:src/test/resources/schema-parallel-history.dot"), Segment("ph01right@ParallelHistory:src/test/resources/schema-parallel-history.dot")))))

  public val ph01leftRegionId: RegionId
    get() = regions[0]

  public val ph01rightRegionId: RegionId
    get() = regions[1]

  override fun target(
    regionId: RegionId,
    segment: Segment,
    rootSegmentAlias: Segment?,
  ): Path? = when (regionId) {
    regions[0] -> {
      val rootSegment = rootSegmentAlias ?: Segment("ph01tabs@ParallelHistory:src/test/resources/schema-parallel-history.dot")
      when(segment.id) {
        "ph01tabs@ParallelHistory:src/test/resources/schema-parallel-history.dot" -> Path(rootSegment)
        "ph01right@ParallelHistory:src/test/resources/schema-parallel-history.dot" -> Path(listOf(rootSegment, Segment("ph01right@ParallelHistory:src/test/resources/schema-parallel-history.dot")))
        "ph01rightScreen@ParallelHistory:src/test/resources/schema-parallel-history.dot" -> Path(listOf(rootSegment, Segment("ph01right@ParallelHistory:src/test/resources/schema-parallel-history.dot"), Segment("ph01rightScreen@ParallelHistory:src/test/resources/schema-parallel-history.dot")))
        "ph01left@ParallelHistory:src/test/resources/schema-parallel-history.dot" -> Path(listOf(rootSegment, Segment("ph01left@ParallelHistory:src/test/resources/schema-parallel-history.dot")))
        "ph01leftScreen@ParallelHistory:src/test/resources/schema-parallel-history.dot" -> Path(listOf(rootSegment, Segment("ph01left@ParallelHistory:src/test/resources/schema-parallel-history.dot"), Segment("ph01leftScreen@ParallelHistory:src/test/resources/schema-parallel-history.dot")))
        else -> null
      }
    }
    regions[1] -> {
      val rootSegment = rootSegmentAlias ?: Segment("ph01tabs@ParallelHistory:src/test/resources/schema-parallel-history.dot")
      when(segment.id) {
        "ph01tabs@ParallelHistory:src/test/resources/schema-parallel-history.dot" -> Path(rootSegment)
        "ph01right@ParallelHistory:src/test/resources/schema-parallel-history.dot" -> Path(listOf(rootSegment, Segment("ph01right@ParallelHistory:src/test/resources/schema-parallel-history.dot")))
        "ph01rightScreen@ParallelHistory:src/test/resources/schema-parallel-history.dot" -> Path(listOf(rootSegment, Segment("ph01right@ParallelHistory:src/test/resources/schema-parallel-history.dot"), Segment("ph01rightScreen@ParallelHistory:src/test/resources/schema-parallel-history.dot")))
        "ph01left@ParallelHistory:src/test/resources/schema-parallel-history.dot" -> Path(listOf(rootSegment, Segment("ph01left@ParallelHistory:src/test/resources/schema-parallel-history.dot")))
        "ph01leftScreen@ParallelHistory:src/test/resources/schema-parallel-history.dot" -> Path(listOf(rootSegment, Segment("ph01left@ParallelHistory:src/test/resources/schema-parallel-history.dot"), Segment("ph01leftScreen@ParallelHistory:src/test/resources/schema-parallel-history.dot")))
        else -> null
      }
    }
    else -> {
      error("""unknown regionId=$regionId""")
    }
  }

  override fun nodeType(
    regionId: RegionId,
    path: Path,
    rootSegmentAlias: Segment?,
  ): Schema.NodeType = when (regionId) {
    regions[0] -> {
      val rootSegment = rootSegmentAlias ?: Segment("ph01tabs@ParallelHistory:src/test/resources/schema-parallel-history.dot")
      when {
        path == Path(rootSegment) -> Schema.NodeType.ParallelFlow
        path == Path(listOf(rootSegment, Segment("ph01right@ParallelHistory:src/test/resources/schema-parallel-history.dot"))) -> Schema.NodeType.Flow
        path == Path(listOf(rootSegment, Segment("ph01right@ParallelHistory:src/test/resources/schema-parallel-history.dot"), Segment("ph01rightScreen@ParallelHistory:src/test/resources/schema-parallel-history.dot"))) -> Schema.NodeType.Screen
        path == Path(listOf(rootSegment, Segment("ph01left@ParallelHistory:src/test/resources/schema-parallel-history.dot"))) -> Schema.NodeType.Flow
        path == Path(listOf(rootSegment, Segment("ph01left@ParallelHistory:src/test/resources/schema-parallel-history.dot"), Segment("ph01leftScreen@ParallelHistory:src/test/resources/schema-parallel-history.dot"))) -> Schema.NodeType.Screen
        else -> {
          error("""internal error: no nodeType for path=$path""")
        }
      }
    }
    regions[1] -> {
      val rootSegment = rootSegmentAlias ?: Segment("ph01tabs@ParallelHistory:src/test/resources/schema-parallel-history.dot")
      when {
        path == Path(rootSegment) -> Schema.NodeType.ParallelFlow
        path == Path(listOf(rootSegment, Segment("ph01right@ParallelHistory:src/test/resources/schema-parallel-history.dot"))) -> Schema.NodeType.Flow
        path == Path(listOf(rootSegment, Segment("ph01right@ParallelHistory:src/test/resources/schema-parallel-history.dot"), Segment("ph01rightScreen@ParallelHistory:src/test/resources/schema-parallel-history.dot"))) -> Schema.NodeType.Screen
        path == Path(listOf(rootSegment, Segment("ph01left@ParallelHistory:src/test/resources/schema-parallel-history.dot"))) -> Schema.NodeType.Flow
        path == Path(listOf(rootSegment, Segment("ph01left@ParallelHistory:src/test/resources/schema-parallel-history.dot"), Segment("ph01leftScreen@ParallelHistory:src/test/resources/schema-parallel-history.dot"))) -> Schema.NodeType.Screen
        else -> {
          error("""internal error: no nodeType for path=$path""")
        }
      }
    }
    else -> {
      error("""unknown regionId=$regionId""")
    }
  }

  override fun isParameterized(
    regionId: RegionId,
    path: Path,
    rootSegmentAlias: Segment?,
  ): Boolean = false

  override fun createChildFlowFinishRequestEvent(
    regionId: RegionId,
    path: Path,
    result: Any,
  ): Event = when (regionId) {
    regions[0] -> {
      when(path) {
        Path(Segment("ph01left@ParallelHistory:src/test/resources/schema-parallel-history.dot")) -> Ph01tabsChildFinishRequest.Ph01left
        else -> {
          error("""internal error: failed to build child finish event for path=$path""")
        }
      }
    }
    regions[1] -> {
      when(path) {
        Path(Segment("ph01right@ParallelHistory:src/test/resources/schema-parallel-history.dot")) -> Ph01tabsChildFinishRequest.Ph01right
        else -> {
          error("""internal error: failed to build child finish event for path=$path""")
        }
      }
    }
    else -> {
      error("""unknown regionId=$regionId""")
    }
  }
}
