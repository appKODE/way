package ru.kode.test.app.schema

import kotlin.Any
import kotlin.Suppress
import kotlin.Unit
import kotlin.collections.Map
import kotlin.collections.Set
import ru.kode.way.FlowNode
import ru.kode.way.MissingPayloadException
import ru.kode.way.Node
import ru.kode.way.NodeBuilder
import ru.kode.way.NullPayload
import ru.kode.way.Path
import ru.kode.way.ScreenNode
import ru.kode.way.Segment
import ru.kode.way.firstSegment

public class Nlf01settingsNodeBuilder(
  private val nodeFactory: Factory,
  override val schema: NestedLocalFlowSchema,
) : NodeBuilder {
  override fun build(
    path: Path,
    payloads: Map<Path, Any>,
    rootSegmentAlias: Segment?,
  ): Node {
    val rootPath = rootSegmentAlias?.let { Path(it) } ?: Path(Segment("nlf01settings@NestedLocalFlow:src/test/resources/node-builders-nested-local-flow.dot"))
    check(path.firstSegment().id == rootPath.firstSegment().id) {
      """illegal path build requested for "nlf01settings" node: $path"""
    }
    return when {
      path == rootPath -> nodeFactory.createRootNode()
      path == targetOrError(Segment("nlf01general@NestedLocalFlow:src/test/resources/node-builders-nested-local-flow.dot"), rootSegmentAlias) -> nodeFactory.createNlf01generalNode()
      else -> error("""illegal path build requested for "nlf01settings" node: $path""")
    }
  }

  override fun invalidateCache(alivePaths: Set<Path>): Unit = Unit

  public fun targetOrError(segment: Segment, rootSegmentAlias: Segment?): Path {
    val path = schema.regions.firstNotNullOfOrNull { schema.target(it, segment, null) } ?: error("""internal error: no target generated for segment "${segment.id}"""")
    return Path(listOf(rootSegmentAlias ?: Segment("nlf01settings@NestedLocalFlow:src/test/resources/node-builders-nested-local-flow.dot")) + path.segments.drop(2))
  }

  @Suppress("UNCHECKED_CAST")
  public fun <T> payloadOrError(
    segment: Segment,
    payloads: Map<Path, Any>,
    rootSegmentAlias: Segment?,
  ): T {
    val targetPath = targetOrError(segment, rootSegmentAlias)
    val payload = payloads[targetPath] ?: throw MissingPayloadException(targetPath)
    return (if (payload === NullPayload) null else payload) as T
  }

  public interface Factory {
    public fun createRootNode(): FlowNode<*>

    public fun createNlf01generalNode(): ScreenNode
  }
}
