package ru.kode.test.app.schema

import kotlin.Any
import kotlin.Boolean
import kotlin.collections.List
import kotlin.collections.Map
import ru.kode.way.Event
import ru.kode.way.Path
import ru.kode.way.RegionId
import ru.kode.way.Schema
import ru.kode.way.Segment

public class PermSchema : Schema {
  override val rootSegment: Segment =
      Segment("perm@Perm:src/test/resources/schema-nullable-imported-root.dot")

  override val childSchemas: Map<Segment, Schema> = emptyMap()

  override val regions: List<RegionId> =
      listOf(RegionId(Path(listOf(Segment("perm@Perm:src/test/resources/schema-nullable-imported-root.dot")))))

  public val permRegionId: RegionId
    get() = regions[0]

  override fun target(
    regionId: RegionId,
    segment: Segment,
    rootSegmentAlias: Segment?,
  ): Path? = when (regionId) {
    regions[0] -> {
      val rootSegment = rootSegmentAlias ?: Segment("perm@Perm:src/test/resources/schema-nullable-imported-root.dot")
      when(segment.id) {
        "perm@Perm:src/test/resources/schema-nullable-imported-root.dot" -> Path(rootSegment)
        "intro@Perm:src/test/resources/schema-nullable-imported-root.dot" -> Path(listOf(rootSegment, Segment("intro@Perm:src/test/resources/schema-nullable-imported-root.dot")))
        "details@Perm:src/test/resources/schema-nullable-imported-root.dot" -> Path(listOf(rootSegment, Segment("intro@Perm:src/test/resources/schema-nullable-imported-root.dot"), Segment("details@Perm:src/test/resources/schema-nullable-imported-root.dot")))
        else -> null
      }
    }
    else -> {
      error("""unknown regionId=$regionId""")
    }
  }

  override fun nodeType(
    regionId: RegionId,
    path: Path,
    rootSegmentAlias: Segment?,
  ): Schema.NodeType = when (regionId) {
    regions[0] -> {
      val rootSegment = rootSegmentAlias ?: Segment("perm@Perm:src/test/resources/schema-nullable-imported-root.dot")
      when {
        path == Path(rootSegment) -> Schema.NodeType.Flow
        path == Path(listOf(rootSegment, Segment("intro@Perm:src/test/resources/schema-nullable-imported-root.dot"))) -> Schema.NodeType.Screen
        path == Path(listOf(rootSegment, Segment("intro@Perm:src/test/resources/schema-nullable-imported-root.dot"), Segment("details@Perm:src/test/resources/schema-nullable-imported-root.dot"))) -> Schema.NodeType.Screen
        else -> {
          error("""internal error: no nodeType for path=$path""")
        }
      }
    }
    else -> {
      error("""unknown regionId=$regionId""")
    }
  }

  override fun isParameterized(
    regionId: RegionId,
    path: Path,
    rootSegmentAlias: Segment?,
  ): Boolean {
    val rootSegment = rootSegmentAlias ?: Segment("perm@Perm:src/test/resources/schema-nullable-imported-root.dot")
    return when {
      path == Path(listOf(rootSegment, Segment("intro@Perm:src/test/resources/schema-nullable-imported-root.dot"), Segment("details@Perm:src/test/resources/schema-nullable-imported-root.dot"))) -> true
      else -> false
    }
  }

  override fun createChildFlowFinishRequestEvent(
    regionId: RegionId,
    path: Path,
    result: Any,
  ): Event = when (regionId) {
    regions[0] -> {
      when(path) {
        else -> {
          error("""internal error: failed to build child finish event for path=$path""")
        }
      }
    }
    else -> {
      error("""unknown regionId=$regionId""")
    }
  }
}
