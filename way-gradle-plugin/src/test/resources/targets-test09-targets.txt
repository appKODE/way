package ru.kode.test.app.schema

import kotlin.String
import ru.kode.way.FlowTarget
import ru.kode.way.HistoryTarget
import ru.kode.way.NullPayload
import ru.kode.way.Path
import ru.kode.way.ScreenTarget
import ru.kode.way.Segment
import ru.kode.way.Target
import ru.kode.way.append

public class AppTargets(
  private val prefix: Path? = null,
) {
  /**
   * Short target: parameterized ancestors keep their current payloads. Use it inside a flow which is alive; if a parameterized ancestor is not alive anymore, the event is dropped.
   */
  public val cart: FlowTarget =
      FlowTarget(flowPath(Path(listOf(Segment("shop@TargetsTest09:src/test/resources/targets-test09.dot"), Segment("cart@TargetsTest09:src/test/resources/targets-test09.dot")))))

  /**
   * Short target: parameterized ancestors keep their current payloads. Use it inside a flow which is alive; if a parameterized ancestor is not alive anymore, the event is dropped.
   */
  public val promo: FlowTarget =
      FlowTarget(flowPath(Path(listOf(Segment("shop@TargetsTest09:src/test/resources/targets-test09.dot"), Segment("item@TargetsTest09:src/test/resources/targets-test09.dot"), Segment("promo@TargetsTest09:src/test/resources/targets-test09.dot")))))

  public fun shop(shopId: String?): FlowTarget = FlowTarget(flowPath(Path(listOf(Segment("shop@TargetsTest09:src/test/resources/targets-test09.dot")))), payload = shopId ?: NullPayload)

  /**
   * Full target: passes payloads for every parameterized ancestor, so they can be rebuilt. Use it for a cold start, a flow's `initial`, an `AbsoluteTarget` or to jump into a branch which is not alive.
   */
  public fun cart(shopId: String?): FlowTarget = FlowTarget(flowPath(Path(listOf(Segment("shop@TargetsTest09:src/test/resources/targets-test09.dot"), Segment("cart@TargetsTest09:src/test/resources/targets-test09.dot")))), ancestorPayloads = mapOf(Segment("shop@TargetsTest09:src/test/resources/targets-test09.dot") to shopId))

  /**
   * Full target: passes payloads for every parameterized ancestor, so they can be rebuilt. Use it for a cold start, a flow's `initial`, an `AbsoluteTarget` or to jump into a branch which is not alive.
   */
  public fun promo(shopId: String?, `object`: String): FlowTarget = FlowTarget(flowPath(Path(listOf(Segment("shop@TargetsTest09:src/test/resources/targets-test09.dot"), Segment("item@TargetsTest09:src/test/resources/targets-test09.dot"), Segment("promo@TargetsTest09:src/test/resources/targets-test09.dot")))), ancestorPayloads = mapOf(Segment("shop@TargetsTest09:src/test/resources/targets-test09.dot") to shopId, Segment("item@TargetsTest09:src/test/resources/targets-test09.dot") to `object`))

  private fun flowPath(path: Path): Path = prefix?.append(path) ?: path
}
