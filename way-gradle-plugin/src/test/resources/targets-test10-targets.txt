package ru.kode.test.app.schema

import kotlin.Int
import kotlin.String
import ru.kode.way.FlowTarget
import ru.kode.way.Path
import ru.kode.way.ScreenTarget
import ru.kode.way.Segment
import ru.kode.way.Target
import ru.kode.way.append

public class AppTargets(
  private val prefix: Path? = null,
) {
  /**
   * Short target: parameterized ancestors keep their current payloads. Use it inside a flow which is alive; if a parameterized ancestor is not alive anymore, the event is dropped.
   */
  public val tabA: FlowTarget =
      FlowTarget(flowPath(Path(listOf(Segment("tabs@TargetsTest10:src/test/resources/targets-test10.dot"), Segment("tabA@TargetsTest10:src/test/resources/targets-test10.dot")))))

  public fun tabs(tabsId: Int): FlowTarget = FlowTarget(flowPath(Path(listOf(Segment("tabs@TargetsTest10:src/test/resources/targets-test10.dot")))), payload = tabsId)

  /**
   * Short target: parameterized ancestors keep their current payloads. Use it inside a flow which is alive; if a parameterized ancestor is not alive anymore, the event is dropped.
   */
  public fun tabB(bId: String): FlowTarget = FlowTarget(flowPath(Path(listOf(Segment("tabs@TargetsTest10:src/test/resources/targets-test10.dot"), Segment("tabB@TargetsTest10:src/test/resources/targets-test10.dot")))), payload = bId)

  /**
   * Full target: passes payloads for every parameterized ancestor, so they can be rebuilt. Use it for a cold start, a flow's `initial`, an `AbsoluteTarget` or to jump into a branch which is not alive.
   */
  public fun tabB(tabsId: Int, bId: String): FlowTarget = FlowTarget(flowPath(Path(listOf(Segment("tabs@TargetsTest10:src/test/resources/targets-test10.dot"), Segment("tabB@TargetsTest10:src/test/resources/targets-test10.dot")))), payload = bId, ancestorPayloads = mapOf(Segment("tabs@TargetsTest10:src/test/resources/targets-test10.dot") to tabsId))

  /**
   * Full target: passes payloads for every parameterized ancestor, so they can be rebuilt. Use it for a cold start, a flow's `initial`, an `AbsoluteTarget` or to jump into a branch which is not alive.
   */
  public fun tabA(tabsId: Int): FlowTarget = FlowTarget(flowPath(Path(listOf(Segment("tabs@TargetsTest10:src/test/resources/targets-test10.dot"), Segment("tabA@TargetsTest10:src/test/resources/targets-test10.dot")))), ancestorPayloads = mapOf(Segment("tabs@TargetsTest10:src/test/resources/targets-test10.dot") to tabsId))

  private fun flowPath(path: Path): Path = prefix?.append(path) ?: path
}

public class TabATargets(
  private val prefix: Path? = null,
) {
  public val screenA: ScreenTarget =
      ScreenTarget(flowPath(Path(listOf(Segment("screenA@TargetsTest10:src/test/resources/targets-test10.dot")))))

  private fun flowPath(path: Path): Path = prefix?.append(path) ?: path
}

public class TabBTargets(
  private val prefix: Path? = null,
) {
  public fun screenB(page: Int): ScreenTarget = ScreenTarget(flowPath(Path(listOf(Segment("screenB@TargetsTest10:src/test/resources/targets-test10.dot")))), payload = page)

  private fun flowPath(path: Path): Path = prefix?.append(path) ?: path
}

public val Target.Companion.app: AppTargets
  get() = AppTargets()

public val Target.Companion.tabB: TabBTargets
  get() = TabBTargets()

public val Target.Companion.tabA: TabATargets
  get() = TabATargets()
