package ru.kode.test.app.schema

import kotlin.Any
import kotlin.Boolean
import kotlin.collections.List
import kotlin.collections.Map
import ru.kode.way.Event
import ru.kode.way.Path
import ru.kode.way.RegionId
import ru.kode.way.Schema
import ru.kode.way.Segment

public class TargetsTest04Schema(
  private val permissionsSchema: Schema,
) : Schema {
  override val rootSegment: Segment =
      Segment("app@TargetsTest04:src/test/resources/targets-test04.dot")

  override val childSchemas: Map<Segment, Schema> =
      mapOf(Segment("permissions@TargetsTest04:src/test/resources/targets-test04.dot") to permissionsSchema)

  override val regions: List<RegionId> =
      listOf(RegionId(Path(listOf(Segment("app@TargetsTest04:src/test/resources/targets-test04.dot")))))

  public val appRegionId: RegionId
    get() = regions[0]

  override fun target(
    regionId: RegionId,
    segment: Segment,
    rootSegmentAlias: Segment?,
  ): Path? = when (regionId) {
    regions[0] -> {
      val rootSegment = rootSegmentAlias ?: Segment("app@TargetsTest04:src/test/resources/targets-test04.dot")
      when(segment.id) {
        "app@TargetsTest04:src/test/resources/targets-test04.dot" -> Path(rootSegment)
        "main@TargetsTest04:src/test/resources/targets-test04.dot" -> Path(listOf(rootSegment, Segment("main@TargetsTest04:src/test/resources/targets-test04.dot")))
        "intro@TargetsTest04:src/test/resources/targets-test04.dot" -> Path(listOf(rootSegment, Segment("intro@TargetsTest04:src/test/resources/targets-test04.dot")))
        "page1@TargetsTest04:src/test/resources/targets-test04.dot" -> Path(listOf(rootSegment, Segment("intro@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page1@TargetsTest04:src/test/resources/targets-test04.dot")))
        "page2@TargetsTest04:src/test/resources/targets-test04.dot" -> Path(listOf(rootSegment, Segment("intro@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page1@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page2@TargetsTest04:src/test/resources/targets-test04.dot")))
        "permissions@TargetsTest04:src/test/resources/targets-test04.dot" -> Path(listOf(rootSegment, Segment("intro@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page1@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page2@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("permissions@TargetsTest04:src/test/resources/targets-test04.dot")))
        else -> null
      }
    }
    else -> {
      error("""unknown regionId=$regionId""")
    }
  }

  override fun nodeType(
    regionId: RegionId,
    path: Path,
    rootSegmentAlias: Segment?,
  ): Schema.NodeType = when (regionId) {
    regions[0] -> {
      val rootSegment = rootSegmentAlias ?: Segment("app@TargetsTest04:src/test/resources/targets-test04.dot")
      when {
        path == Path(rootSegment) -> Schema.NodeType.Flow
        path == Path(listOf(rootSegment, Segment("main@TargetsTest04:src/test/resources/targets-test04.dot"))) -> Schema.NodeType.Flow
        path == Path(listOf(rootSegment, Segment("intro@TargetsTest04:src/test/resources/targets-test04.dot"))) -> Schema.NodeType.Screen
        path == Path(listOf(rootSegment, Segment("intro@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page1@TargetsTest04:src/test/resources/targets-test04.dot"))) -> Schema.NodeType.Screen
        path == Path(listOf(rootSegment, Segment("intro@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page1@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page2@TargetsTest04:src/test/resources/targets-test04.dot"))) -> Schema.NodeType.Screen
        path == Path(listOf(rootSegment, Segment("intro@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page1@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page2@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("permissions@TargetsTest04:src/test/resources/targets-test04.dot"))) -> Schema.NodeType.Flow
        else -> {
          error("""internal error: no nodeType for path=$path""")
        }
      }
    }
    else -> {
      error("""unknown regionId=$regionId""")
    }
  }

  override fun isParameterized(
    regionId: RegionId,
    path: Path,
    rootSegmentAlias: Segment?,
  ): Boolean {
    val rootSegment = rootSegmentAlias ?: Segment("app@TargetsTest04:src/test/resources/targets-test04.dot")
    return when {
      path == Path(rootSegment) -> true
      path == Path(listOf(rootSegment, Segment("main@TargetsTest04:src/test/resources/targets-test04.dot"))) -> true
      path == Path(listOf(rootSegment, Segment("intro@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page1@TargetsTest04:src/test/resources/targets-test04.dot"))) -> true
      path == Path(listOf(rootSegment, Segment("intro@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page1@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page2@TargetsTest04:src/test/resources/targets-test04.dot"))) -> true
      path == Path(listOf(rootSegment, Segment("intro@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page1@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page2@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("permissions@TargetsTest04:src/test/resources/targets-test04.dot"))) -> true
      else -> false
    }
  }

  override fun createChildFlowFinishRequestEvent(
    regionId: RegionId,
    path: Path,
    result: Any,
  ): Event = when (regionId) {
    regions[0] -> {
      when(path) {
        Path(listOf(rootSegment, Segment("main@TargetsTest04:src/test/resources/targets-test04.dot"))) -> AppChildFinishRequest.Main
        Path(listOf(rootSegment, Segment("intro@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page1@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("page2@TargetsTest04:src/test/resources/targets-test04.dot"), Segment("permissions@TargetsTest04:src/test/resources/targets-test04.dot"))) -> AppChildFinishRequest.Permissions
        else -> {
          error("""internal error: failed to build child finish event for path=$path""")
        }
      }
    }
    else -> {
      error("""unknown regionId=$regionId""")
    }
  }
}
